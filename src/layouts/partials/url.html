{{- /* Base-path-safe URL resolver. Accepts a string, or {url, page}. Markdown references are resolved through Hugo so moves fail the build. Fragments are split before resolution and reattached on every branch. Already-resolved menu/page permalinks must not pass through this partial; use menu-url.html for menu entries. */ -}} {{- $u := "" }}{{ $page := "" -}} {{- if reflect.IsMap . }}{{ $u = .url }}{{ $page = .page }}{{ else }}{{ $u = . }}{{ end -}} {{- $u = string $u -}} {{- if or (urls.Parse $u).IsAbs (hasPrefix $u "//") (hasPrefix $u "#") (hasPrefix $u "mailto:") (hasPrefix $u "tel:") -}} {{- $u -}} {{- else if and $page (or (hasPrefix $u "<") (strings.Contains $u ".md")) -}} {{- $parsed := urls.Parse (trim $u "<>") -}} {{- $target := $parsed.Path -}} {{- with $page.GetPage $target -}} {{- .RelPermalink -}}{{- with $parsed.Fragment }}#{{ . }}{{ end -}} {{- else -}} {{- errorf "url.html: no page found for ref %q on %s" $target $page.Path -}} {{- end -}} {{- else -}} {{- $parsed := urls.Parse $u -}} {{- strings.TrimPrefix "/" $parsed.Path | relLangURL -}}{{- with $parsed.RawQuery }}?{{ . }}{{ end -}}{{- with $parsed.Fragment }}#{{ . }}{{ end -}} {{- end -}}
