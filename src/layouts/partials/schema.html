{{- /* schema.org JSON-LD. TechArticle for documentation, BlogPosting for posts, WebSite plus a breadcrumb trail elsewhere. Kept minimal and truthful — only fields the content actually provides. */ -}} {{- $p := .page }}{{ $desc := .description }}{{ $img := .image -}} {{- $type := "WebPage" -}} {{- if $p.IsHome }}{{ $type = "WebSite" }} {{- else if eq $p.Section "changelog" }}{{ $type = "BlogPosting" }} {{- else if eq $p.Section "docs" }}{{ $type = "TechArticle" }}{{ end -}} {{- $data := dict "@context" "https://schema.org" "@type" $type "name" (cond $p.IsHome site.Title $p.Title) "headline" $p.Title "description" $desc "url" $p.Permalink "inLanguage" $p.Language.Locale -}} {{- with $img }}{{ $data = merge $data (dict "image" .) }}{{ end -}} {{- if $p.Date }}{{ $data = merge $data (dict "datePublished" ($p.Date.Format "2006-01-02")) }}{{ end -}} {{- if $p.Lastmod }}{{ $data = merge $data (dict "dateModified" ($p.Lastmod.Format "2006-01-02")) }}{{ end -}} {{- with $p.Params.author }}{{ $data = merge $data (dict "author" (dict "@type" "Organization" "name" .)) }}{{ end -}} {{- if $p.IsHome }}{{ $data = merge $data (dict "publisher" (dict "@type" "Organization" "name" site.Title "url" site.Home.Permalink)) }}{{ end -}} {{- $crumbs := $p.Ancestors.Reverse -}} {{- if $crumbs -}} {{- $items := slice -}} {{- range $i, $a := $crumbs -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "name" (cond $a.IsHome (i18n "home") $a.LinkTitle) "item" $a.Permalink) -}} {{- end -}} {{- $items = $items | append (dict "@type" "ListItem" "position" (add (len $crumbs) 1) "name" $p.LinkTitle "item" $p.Permalink) -}} {{- end -}}
