{{- $src := .Get "src" | default (.Get 0) -}} {{- if not $src }}{{ errorf "chart shortcode requires a CSV src on %s" .Page.File.Path }}{{ end -}} {{- $resource := or (.Page.Resources.GetMatch $src) (resources.Get $src) -}} {{- if not $resource }}{{ errorf "chart data %q was not found as a page or global asset resource" $src }}{{ end -}} {{- $rows := transform.Unmarshal (dict "delimiter" ",") $resource.Content -}} {{- if lt (len $rows) 2 }}{{ errorf "chart data %q needs a header and at least one row" $src }}{{ end -}} {{- $data := after 1 $rows -}} {{- $kind := lower (.Get "type" | default "bar") -}} {{- if not (in (slice "bar" "line" "dot") $kind) }}{{ errorf "chart type %q must be bar, line, or dot" $kind }}{{ end -}} {{- $title := .Get "title" | default "Chart" -}} {{- $caption := .Get "caption" -}} {{- $xLabel := .Get "x-label" -}} {{- $yLabel := .Get "y-label" -}} {{- $width := 720.0 }}{{ $height := 380.0 -}} {{- $left := 72.0 }}{{ $right := 24.0 }}{{ $top := 42.0 }}{{ $bottom := 72.0 -}} {{- $plotW := sub $width (add $left $right) }}{{ $plotH := sub $height (add $top $bottom) -}} {{- $maxScratch := newScratch }}{{ $maxScratch.Set "value" 0.0 -}} {{- range $data -}} {{- $value := float (index . 1) -}} {{- if gt $value ($maxScratch.Get "value") }}{{ $maxScratch.Set "value" $value }}{{ end -}} {{- end -}} {{- $max := float ($maxScratch.Get "value") -}} {{- if le $max 0.0 }}{{ errorf "chart data %q must contain a positive numeric value in column two" $src }}{{ end -}} {{- $count := len $data }}{{ $slot := div $plotW (float $count) -}} {{- $id := printf "chart-%d" .Ordinal -}} {{- $summary := newScratch }}{{ $summary.Set "value" "" -}} {{- range $index, $row := $data -}} {{- $separator := cond (eq $index 0) "" "; " -}} {{- $summary.Set "value" (printf "%s%s%s: %s" ($summary.Get "value") $separator (index $row 0) (index $row 1)) -}} {{- end -}}
