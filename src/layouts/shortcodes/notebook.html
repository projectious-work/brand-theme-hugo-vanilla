{{- /* Embeds a notebook converted by scripts/notebooks.sh. Lookup order, first match wins, no site-wide scan: 1. a page resource — the idiomatic case, notebook.ipynb beside index.md 2. content/_notebooks/.md — the documented shared location Two notebooks with the same basename used to resolve unpredictably because every page in the site was scanned; both paths are now exact, and a resource glob matching more than one file is an error rather than a guess. */ -}} {{- $name := .Get "name" | default (.Get 0) -}} {{- $md := "" }}{{ $from := "" -}} {{- $matches := .Page.Resources.Match (printf "%s.md" $name) -}} {{- if gt (len $matches) 1 -}} {{- errorf "notebook %q matches %d page resources on %s — rename so the basename is unique" $name (len $matches) .Page.Path -}} {{- end -}} {{- with index $matches 0 }}{{ $md = .Content }}{{ $from = "resource" }}{{ end -}} {{- if not $md -}} {{- with site.GetPage (printf "/_notebooks/%s" $name) }}{{ $md = .RawContent }}{{ $from = "content" }}{{ end -}} {{- end -}} {{- if $md -}}
{{ $md | .Page.RenderString (dict "display" "block") }}
{{- else -}} {{- warnf "notebook %q not found for %s — run scripts/notebooks.sh" $name .Page.Path -}}

{{ i18n "notebookMissing" (dict "Name" $name) }}
{{- end -}}
