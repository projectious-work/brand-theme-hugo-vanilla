{{- /* Markdown images become figures with a caption from the title, lazy loading, and intrinsic dimensions where the format has them. MediaType.MainType == "image" is not enough to read .Width/.Height — an SVG is an image resource without metadata, so reflect.IsImageResourceWithMeta is the guard. A sibling -dark. is swapped by colour mode. */ -}} {{- $src := .Destination -}} {{- $url := $src -}} {{- $dark := "" }}{{ $w := 0 }}{{ $h := 0 -}} {{- if not (urls.Parse $src).IsAbs -}} {{- $res := or (.Page.Resources.GetMatch $src) (resources.Get $src) -}} {{- with $res -}} {{- $url = .RelPermalink -}} {{- if reflect.IsImageResourceWithMeta . }}{{ $w = .Width }}{{ $h = .Height }}{{ end -}} {{- end -}} {{- $ext := path.Ext $src -}} {{- $darkPath := printf "%s-dark%s" (strings.TrimSuffix $ext $src) $ext -}} {{- $dark = or (.Page.Resources.GetMatch $darkPath) (resources.Get $darkPath) -}} {{- end -}}
